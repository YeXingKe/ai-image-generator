### 2) 新建 `lib/auth/session.ts`

```ts
import "server-only";
import { SignJWT, jwtVerify } from "jose";
import { cookies } from "next/headers";

const COOKIE_NAME = "moyu_session";

function getSecretKey() {
  const secret = process.env.AUTH_SECRET;
  if (!secret || secret.length < 16) {
    throw new Error("AUTH_SECRET missing or too short (min 16 chars)");
  }
  return new TextEncoder().encode(secret);
}

export async function createSessionToken(userId: string): Promise<string> {
  return new SignJWT({ sub: userId })
    .setProtectedHeader({ alg: "HS256" })
    .setIssuedAt()
    .setExpirationTime("7d")
    .sign(getSecretKey());
}

export async function verifySessionToken(
  token: string,
): Promise<string | null> {
  try {
    const { payload } = await jwtVerify(token, getSecretKey());
    const sub = payload.sub;
    return typeof sub === "string" ? sub : null;
  } catch {
    return null;
  }
}

/** 在 Server Action / Route Handler 里设置登录 Cookie */
export async function setSessionCookie(userId: string): Promise<void> {
  const token = await createSessionToken(userId);
  const jar = await cookies();
  jar.set(COOKIE_NAME, token, {
    httpOnly: true,
    sameSite: "lax",
    path: "/",
    secure: process.env.NODE_ENV === "production",
    maxAge: 60 * 60 * 24 * 7,
  });
}

export async function clearSessionCookie(): Promise<void> {
  const jar = await cookies();
  jar.delete(COOKIE_NAME);
}

/** 从 Request 的 Cookie 头解析 userId（API Route 专用） */
export async function getUserIdFromRequest(
  request: Request,
): Promise<string | null> {
  const cookieHeader = request.headers.get("cookie") ?? "";
  const match = cookieHeader.match(new RegExp(`${COOKIE_NAME}=([^;]+)`));
  if (!match?.[1]) return null;
  return verifySessionToken(decodeURIComponent(match[1]));
}

/** App Router 服务端组件里读当前用户（可选） */
export async function getUserIdFromCookies(): Promise<string | null> {
  const jar = await cookies();
  const token = jar.get(COOKIE_NAME)?.value;
  if (!token) return null;
  return verifySessionToken(token);
}
```

import { hashPassword } from "@/lib/auth/password";
import { setSessionCookie } from "@/lib/auth/session";
import { prisma } from "@/lib/prisma";

export async function POST(request: Request) {
  let body: unknown;
  try {
    body = await request.json();
  } catch {
    return fail("VALIDATION_ERROR", "Invalid JSON body", 400);
  }

  const email =
    typeof body === "object" && body && "email" in body
      ? String((body as { email: unknown }).email ?? "").trim()
      : "";
  const password =
    typeof body === "object" && body && "password" in body
      ? String((body as { password: unknown }).password ?? "")
      : "";

  if (!email || !email.includes("@")) {
    return fail("VALIDATION_ERROR", "请输入有效邮箱", 400);
  }
  if (password.length < 6) {
    return fail("VALIDATION_ERROR", "密码至少 6 位", 400);
  }

  const exists = await prisma.user.findUnique({ where: { email } });
  if (exists) {
    return fail("VALIDATION_ERROR", "该邮箱已注册", 400);
  }

  const hashed = await hashPassword(password);

  const user = await prisma.$transaction(async (tx) => {
    const created = await tx.user.create({
      data: {
        email,
        password: hashed,
        credits: 10,
      },
    });
    await tx.creditTransaction.create({
      data: {
        userId: created.id,
        amount: 10,
        type: "recharge",
        description: "注册赠送体验积分",
      },
    });
    return created;
  });

  await setSessionCookie(user.id);

  return ok({
    email: user.email,
    credits: user.credits,
  });
}

import { useEffect, useState } from "react";
import { useRouter } from "next/navigation";
import { setSession, type ClientSession } from "@/lib/auth-session";

type AuthGateProps = {
  children: (session: ClientSession) => React.ReactNode;
};

export function AuthGate({ children }: AuthGateProps) {
  const router = useRouter();
  const [session, setLocal] = useState<ClientSession | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let cancelled = false;
    (async () => {
      try {
        const res = await fetch("/api/auth/me", { credentials: "include" });
        const json = (await res.json()) as {
          ok: boolean;
          data?: ClientSession;
        };
        if (!json.ok || !json.data) {
          router.replace("/login");
          return;
        }
        if (!cancelled) {
          setSession(json.data);
          setLocal(json.data);
        }
      } catch {
        router.replace("/login");
      } finally {
        if (!cancelled) setLoading(false);
      }
    })();
    return () => {
      cancelled = true;
    };
  }, [router]);

  if (loading || !session) {
    return (
      <div className="flex min-h-dvh items-center justify-center bg-[var(--bg)] text-[var(--muted)]">
        加载中…
      </div>
    );
  }

  return <>{children(session)}</>;
}

### 4) 覆盖 `app/api/generate/route.ts`（同步版 + 预扣退分）

```ts
import { fail, ok } from "@/lib/api-response";
import { getUserIdFromRequest } from "@/lib/auth/session";
import { calcGenerationCost } from "@/lib/credits";
import { getModelVersion } from "@/lib/models";
import { prisma } from "@/lib/prisma";
import { getProvider } from "@/lib/providers";
import { generateBodySchema } from "@/lib/validators/generate";

export async function POST(request: Request) {
  const userId = await getUserIdFromRequest(request);
  if (!userId) {
    return fail("UNAUTHORIZED", "未登录", 401);
  }

  let json: unknown;
  try {
    json = await request.json();
  } catch {
    return fail("VALIDATION_ERROR", "Invalid JSON body", 400);
  }

  const parsed = generateBodySchema.safeParse(json);
  if (!parsed.success) {
    return fail(
      "VALIDATION_ERROR",
      parsed.error.issues[0]?.message ?? "Invalid body",
      400,
    );
  }

  const body = parsed.data;
  const version = getModelVersion(body.modelVersion);
  if (!version || version.family !== body.model) {
    return fail("VALIDATION_ERROR", "模型版本无效", 400);
  }
  if (body.mode === "image_to_image" && !version.supportsImageToImage) {
    return fail("VALIDATION_ERROR", "该版本不支持图生图", 400);
  }
  if (body.mode === "image_to_image" && !body.refImageUrl) {
    return fail("VALIDATION_ERROR", "图生图需要参考图 URL", 400);
  }

  const cost = calcGenerationCost(body.modelVersion, body.mode);

  // —— 事务：预扣 + 流水 + 任务 ——
  let taskId: string;
  let creditsLeft: number;
  try {
    const result = await prisma.$transaction(async (tx) => {
      const updated = await tx.user.updateMany({
        where: { id: userId, credits: { gte: cost } },
        data: { credits: { decrement: cost } },
      });
      if (updated.count === 0) {
        throw new Error("INSUFFICIENT_CREDITS");
      }

      const user = await tx.user.findUniqueOrThrow({ where: { id: userId } });

      const task = await tx.generationTask.create({
        data: {
          userId,
          model: body.model,
          mode: body.mode,
          prompt: body.prompt,
          refImageUrl: body.refImageUrl ?? null,
          cost,
          status: "processing",
        },
      });

      await tx.creditTransaction.create({
        data: {
          userId,
          amount: -cost,
          type: "generation",
          description: `生图消耗 ${cost} 积分`,
          relatedTaskId: task.id,
        },
      });

      return { taskId: task.id, creditsLeft: user.credits };
    });
    taskId = result.taskId;
    creditsLeft = result.creditsLeft;
  } catch (e) {
    if (e instanceof Error && e.message === "INSUFFICIENT_CREDITS") {
      return fail("INSUFFICIENT_CREDITS", "积分不足，请先充值", 400);
    }
    throw e;
  }

  // —— 调模型（阶段 A 同步）——
  try {
    const provider = getProvider(body.model);
    const { imageUrl } = await provider.generate({
      modelVersion: body.modelVersion,
      mode: body.mode,
      prompt: body.prompt,
      size: body.size,
      refImageUrl: body.refImageUrl,
    });

    await prisma.generationTask.update({
      where: { id: taskId },
      data: { status: "done", resultUrl: imageUrl },
    });

    return ok({
      taskId,
      cost,
      creditsLeft,
      status: "done" as const,
      resultUrl: imageUrl,
    });
  } catch (err) {
    const message = err instanceof Error ? err.message : "生成失败";

    await prisma.$transaction(async (tx) => {
      await tx.user.update({
        where: { id: userId },
        data: { credits: { increment: cost } },
      });
      await tx.creditTransaction.create({
        data: {
          userId,
          amount: cost,
          type: "refund",
          description: `生图失败退还 ${cost} 积分`,
          relatedTaskId: taskId,
        },
      });
      await tx.generationTask.update({
        where: { id: taskId },
        data: { status: "failed", errorMessage: message },
      });
    });

    const user = await prisma.user.findUniqueOrThrow({ where: { id: userId } });

    return fail("GENERATION_FAILED", message, 500);
    // 若希望前端同时拿到 creditsLeft，可改为 ok + status failed；
    // 这里用 fail，前端应再调 /api/auth/me 刷新余额。
  }
}
```

> 注意：上面 `catch` 里 `return fail` 前已退分。前端失败后请再请求 `/api/auth/me`。

import { getUserIdFromRequest } from "@/lib/auth/session";
import { prisma } from "@/lib/prisma";

export async function GET(request: Request) {
  const userId = await getUserIdFromRequest(request);
  if (!userId) {
    return fail("UNAUTHORIZED", "未登录", 401);
  }

  const { searchParams } = new URL(request.url);
  const page = Math.max(1, Number(searchParams.get("page") ?? 1));
  const pageSize = Math.min(50, Math.max(1, Number(searchParams.get("pageSize") ?? 20)));
  const skip = (page - 1) * pageSize;

  const [total, items] = await Promise.all([
    prisma.generationTask.count({ where: { userId } }),
    prisma.generationTask.findMany({
      where: { userId },
      orderBy: { createdAt: "desc" },
      skip,
      take: pageSize,
      select: {
        id: true,
        model: true,
        mode: true,
        prompt: true,
        resultUrl: true,
        cost: true,
        status: true,
        createdAt: true,
      },
    }),
  ]);

  return ok({ page, pageSize, total, items });
}

### 5) 新建 `app/api/recharge/mock-pay/route.ts`（仅开发）

```ts
import { fail, ok } from "@/lib/api-response";
import { getUserIdFromRequest } from "@/lib/auth/session";
import { prisma } from "@/lib/prisma";

export async function POST(request: Request) {
  if (process.env.NODE_ENV === "production") {
    return fail("NOT_FOUND", "Not found", 404);
  }

  const userId = await getUserIdFromRequest(request);
  if (!userId) return fail("UNAUTHORIZED", "未登录", 401);

  const { orderId } = (await request.json()) as { orderId?: string };
  if (!orderId) return fail("VALIDATION_ERROR", "缺少 orderId", 400);

  const order = await prisma.rechargeOrder.findFirst({
    where: { id: orderId, userId },
  });
  if (!order) return fail("NOT_FOUND", "订单不存在", 404);
  if (order.status === "paid") {
    const user = await prisma.user.findUniqueOrThrow({ where: { id: userId } });
    return ok({ credits: user.credits, alreadyPaid: true });
  }

  const user = await prisma.$transaction(async (tx) => {
    const updated = await tx.rechargeOrder.updateMany({
      where: { id: orderId, status: "pending" },
      data: { status: "paid", paidAt: new Date(), providerTradeNo: `mock_${Date.now()}` },
    });
    if (updated.count === 0) {
      return tx.user.findUniqueOrThrow({ where: { id: userId } });
    }
    await tx.user.update({
      where: { id: userId },
      data: { credits: { increment: order.credits } },
    });
    await tx.creditTransaction.create({
      data: {
        userId,
        amount: order.credits,
        type: "recharge",
        description: `充值套餐 ${order.packageId}`,
      },
    });
    return tx.user.findUniqueOrThrow({ where: { id: userId } });
  });

  return ok({ credits: user.credits, alreadyPaid: false });
}
```
